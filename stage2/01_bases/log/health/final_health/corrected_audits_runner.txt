import datetime,hashlib,json,os,pathlib,re,subprocess,time,sys
root=pathlib.Path('/home/chang/arc');out=root/'.scratch/stage2-bases/runtime/final-health'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
now=lambda:datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=8))).isoformat()
path=out/'corrected_audits.json'
if path.exists():raise SystemExit('Existing corrected audits; do not silently repeat')
if subprocess.check_output(['git','status','--porcelain'],cwd=root):raise SystemExit('Canonical dev must be clean')
checks=['check_paths','check_tracked'];pins={n:sha(root/'script'/f'{n}.sh') for n in checks}
v={'schema':'stage2-failclosed-real-canonical-audits-v1','status':'running','started_at':now(),'checked_commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'cwd':str(root),'PATH':os.environ['PATH'],'script_pins':pins,'checks':[],'verify_all_repeated':False}
for n in checks:
 stdout=out/(n+'_corrected.stdout.log');stderr=out/(n+'_corrected.stderr.log');resources=out/(n+'_corrected.time.txt')
 argv=['/usr/bin/time','-v','-o',str(resources),'/bin/bash',str(root/'script'/f'{n}.sh')]
 started=now();clock=time.monotonic()
 with stdout.open('wb') as a,stderr.open('wb') as b:r=subprocess.run(argv,cwd=root,stdout=a,stderr=b)
 text=stdout.read_text();error=stderr.read_text()
 scan=re.search(r'tracked text files scanned: (\d+)' if n=='check_paths' else r'scan: (\d+) code files on disk',text)
 valid=r.returncode==0 and scan is not None and int(scan[1])>0 and not error and 'RESULT: PASS' in text and not re.search(r'Segmentation fault|core dumped|Broken pipe',text,re.I)
 v['checks'].append({'name':n,'argv':argv,'started_at':started,'finished_at':now(),'elapsed_seconds':time.monotonic()-clock,'exit_code':r.returncode,'scan_files':int(scan[1]) if scan else None,'stderr_bytes':stderr.stat().st_size,'valid':valid,'stdout':str(stdout),'stdout_sha256':sha(stdout),'stderr':str(stderr),'stderr_sha256':sha(stderr),'resources':str(resources),'resources_sha256':sha(resources)})
 path.write_text(json.dumps(v,indent=2,sort_keys=True)+'\n');print(n+' exit='+str(r.returncode)+' actualscan='+str(int(scan[1]) if scan else None)+' valid='+str(valid),flush=True)
v['finished_at']=now();v['script_pins_unchanged']=all(sha(root/'script'/f'{n}.sh')==pins[n] for n in checks);v['canonical_status']=subprocess.check_output(['git','status','--porcelain'],cwd=root,text=True);v['status']='PASS' if all(x['valid'] for x in v['checks']) and v['script_pins_unchanged'] and not v['canonical_status'] else 'FAIL'
path.write_text(json.dumps(v,indent=2,sort_keys=True)+'\n');print('Corrected canonical audits '+v['status'],flush=True)
sys.exit(0 if v['status']=='PASS' else 1)
