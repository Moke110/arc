import os
os.environ['JAX_PLATFORM_NAME']='cuda'
os.environ['OMP_NUM_THREADS']='2'
os.environ['OPENBLAS_NUM_THREADS']='2'
os.environ['MKL_NUM_THREADS']='2'
os.environ['XLA_FLAGS']='--xla_cpu_multi_thread_eigen=false intra_op_parallelism_threads=2 --xla_gpu_cuda_data_dir=/home/chang/arc/pert_model/CMonge/local/tenv_cuda/nvidia/cuda_nvcc'
os.environ['PATH']='/home/chang/arc/pert_model/CMonge/local/tenv_cuda/nvidia/cuda_nvcc/bin'+os.pathsep+os.environ.get('PATH','')
os.environ['XLA_PYTHON_CLIENT_PREALLOCATE']='false'
import sys,json,time
from pathlib import Path
sys.dont_write_bytecode=True
root=Path('/home/chang/arc')
sys.path.insert(0,str(root/'stage2/03_modes_tenv/execution/script'))
sys.path.insert(0,str(root/'pert_model/CMonge/local/tenv_cuda'))
from cmonge_adapter import AdaptedTransport,jax,jnp,np,serialization,rows
run=root/'stage2/03_modes_tenv/output/formal/runs/cmonge/fold_0/seed_17'
request=json.loads((run/'request.json').read_text()); files=request['predictor_files']; seed=request['identity']['seed']; rng=np.random.default_rng(seed)
batch=512; genes=request['n_genes']; cases=rows(files['train_cases']); groups=sorted({r['group_id'] for r in cases}); indices=rng.choice([i for i,r in enumerate(cases) if r['group_id']==groups[0]],size=batch,replace=True); selected=[cases[i] for i in indices]
controls=np.load(files['controls'],mmap_mode='r'); morgan=np.load(files['morgan'],mmap_mode='r'); drug_index={r['drug_key']:int(r['morgan_index']) for r in rows(files['drugs'])}
x=jnp.asarray(np.array([controls[int(r['control_index'])] for r in selected],dtype='float32')); c=jnp.asarray(np.array([morgan[drug_index[r['drug_key']]] for r in selected],dtype='float32')); d=jnp.asarray(np.log(np.array([float(r['dose_um']) for r in selected],dtype='float32'))[:,None]); model=AdaptedTransport(genes); params=model.init(jax.random.PRNGKey(seed),x,c,d)['params']; original=np.asarray(params['morgan_adapter']['kernel']).copy(); blob=(run/'best.msgpack').read_bytes(); best=serialization.msgpack_restore(blob); restored=serialization.from_bytes(params,blob)
predict=jax.jit(lambda p,x,c,d:model.apply({'params':p},x,c,d))
before=np.asarray(predict(best,x,c,d)); after=np.asarray(predict(restored,x,c,d)); xb=np.asarray(predict(restored,x+.1,c,d)); xd=np.asarray(predict(restored,x,c,d+1.))
values={'backend':jax.default_backend(),'adapter_parameter_change':float(np.linalg.norm(np.asarray(restored['morgan_adapter']['kernel'])-original)),'baseline_sensitivity':float(np.max(np.abs(xb-after))),'dose_sensitivity':float(np.max(np.abs(xd-after))),'reload_max_absolute_error':float(np.max(np.abs(before-after))),'prediction_min':float(after.min()),'prediction_max':float(after.max()),'prediction_finite':bool(np.isfinite(after).all()),'first_group':groups[0],'first_group_wells':len(set(indices.tolist())),'batch_size':batch}
values['gate_pass']=values['adapter_parameter_change']>0 and values['baseline_sensitivity']>0 and values['dose_sensitivity']>0 and values['reload_max_absolute_error']==0
print(json.dumps(values,indent=2),flush=True)
Path('/tmp/tenv_cmonge_gate_probe_gpu.json').write_text(json.dumps(values,indent=2)+'\n')
assert values['gate_pass'],'adapter/baseline/dose/reload behavior check failed'
